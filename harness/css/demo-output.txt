-- why
   aside, blockquote, #sidebar, .unused, .grid-only, .no-print: no element matches.
   @supports (display: grid) has no rule left, so it goes; @media print keeps
   "aside, nav" whole, because nav is on the page (a rule is kept or dropped as one).

   selectors are always kept (the brief's conservative rule).
   Pseudo-classes and attribute selectors (a:hover, th[data-num],
   label:has(input[type="checkbox"])) count as satisfied.

== Demo 4: invalid input — conditions, use-value, skip-invalid
========================================================================

-- (make-declaration "color" "red;}body{x:y") with no handler that fixes it
   condition: CSS-PROTOCOL:CSS-INVALID-VALUE
   report:    Invalid CSS value: ; outside strings and brackets: "red;}body{x:y"
   input:     "red;}body{x:y"
   a css-error: T, a css-syntax-error: T
   restarts offered at the signal:
     USE-VALUE — Use another value (it is validated again).
     CSS-PROTOCOL:SKIP-INVALID — Skip this item (the constructor returns NIL).

-- use-value: a handler supplies a replacement (validated again)
   inputs refused, in order: ("red;}body{x:y" "red;")
   result: p{color:crimson}

-- skip-invalid: the bad item is dropped, its container keeps the rest
   skipping CSS-PROTOCOL:CSS-INVALID-VALUE: "red;}body{x:y"
   skipping CSS-PROTOCOL:CSS-INVALID-SELECTOR: "p { } body"
   skipping CSS-PROTOCOL:CSS-INVALID-PROPERTY: "col or"
   skipping CSS-PROTOCOL:CSS-INVALID-PRELUDE: "screen {"
   skipping CSS-PROTOCOL:CSS-INVALID-VALUE: "</style><script>"
   stylesheet left: p{margin:0}em{font-style:italic}strong{}
   (rule "p { } body" is NIL: a rule whose selector list is skipped is not a rule;
    @media "screen {" is NIL the same way; "strong" keeps no declarations)

-- each refusal, one by one
   value:               CSS-PROTOCOL:CSS-INVALID-VALUE  "red;}body{x:y"
   value with </:       CSS-PROTOCOL:CSS-INVALID-VALUE  "\"</style>\""
   property:            CSS-PROTOCOL:CSS-INVALID-PROPERTY  "col or"
   selector:            CSS-PROTOCOL:CSS-INVALID-SELECTOR  "p { } body"
   selector (nesting):  CSS-PROTOCOL:CSS-INVALID-SELECTOR  "& > a"
   media query:         CSS-PROTOCOL:CSS-INVALID-PRELUDE  "screen {"
   keyframes name:      CSS-PROTOCOL:CSS-INVALID-IDENTIFIER  ""
   keyframe selector:   CSS-PROTOCOL:CSS-INVALID-SELECTOR  "halfway"

-- text that is data, not syntax, is escaped instead
   (css-string "</style>")  => "\3c /style>"
   (css-ident "2col")         => \32 col
   (css-url "a b\".png")      => url("a b\".png")
   rule with content: (css-string ...): q::before{content:"\3c /style>"}

-- compile-stylesheet with no backend: css-no-backend, then use-value
   condition: CSS-PROTOCOL:CSS-NO-BACKEND
   report:    No CSS backend to compile (:LASS (HARNESS-DEMOS-CSS::BODY :MARGIN 0)); bind CSS-PROTOCOL:*CSS-BACKEND* or pass :backend
   restarts: (USE-VALUE CSS-PROTOCOL:USE-BACKEND)
   result: body{margin:0} (the fallback, EQ: T)

-- how the page switches
   The rules never name a colour, only var(--…):
     body{margin:0;background:var(--bg);color:var(--fg);font-family:system-ui, -apple-system, "Segoe UI", Roboto, "Helvetica Neue", Arial, sans-serif;line-height:1.6;text-rendering:optimizeLegibility}
   OS light, no data-theme      -> :root                       (light values)
   OS dark,  no data-theme      -> :root:not([data-theme="light"]) in @media (dark values)
   data-theme="dark" on <html>  -> :root[data-theme="dark"]    (dark, whatever the OS)
   data-theme="light" on <html> -> :root[data-theme="light"]   (light: the @media rule no longer matches)
   color-scheme: light dark on :root also switches form controls and scrollbars.
