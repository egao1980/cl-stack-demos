--- 2a. in-memory store: two branches, a tag, walks, merge-base, as-of
    refs:
      feature movable -> f2-owner

    walk-revisions (main feature) :topo:

      m2-bump  parents ("m1-init")  recorded 200  "bump to 2"  state (:APP "demo" :VERSION 2 :FLAGS NIL)

    walk feature :until main => ("f1-beta" "f2-owner")
    walk main :footprint ("sexp:/version") => ("m1-init" "m2-bump" "m3-bump")

    resolve id prefix "f2-o" => f2-owner
